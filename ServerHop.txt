local function fetchAndDisplayServers()
    for _, child in ipairs(ListContainer:GetChildren()) do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end
    
    StatusLabel.Text = "Buscando servidores estables..."
    StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 100)
    
    local servers = {}
    local success, raw = pcall(function()
        return game:HttpGet("https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100")
    end)
    
    if success and raw then
        local decodeSuccess, data = pcall(function()
            return HttpService:JSONDecode(raw)
        end)
        if decodeSuccess and data and data.data then
            for _, s in ipairs(data.data) do
                if s.id ~= game.JobId and type(s.playing) == "number" and type(s.maxPlayers) == "number" then
                    -- Asegurar que tenga espacio real
                    if s.playing < s.maxPlayers then
                        table.insert(servers, s)
                    end
                end
            end
        end
    end
    
    table.sort(servers, function(a, b)
        return a.playing < b.playing
    end)
    
    -- Opcional: si hay suficientes servidores, eliminamos el primero porque suele estar saturado
    if #servers > 3 then
        table.remove(servers, 1)
    end
    
    if #servers > 0 then
        StatusLabel.Text = "¡Lista cargada! Prueba del segundo en adelante."
        StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 150)
        
        for i, s in ipairs(servers) do
            local ServerCard = Instance.new("Frame")
            ServerCard.Parent = ListContainer
            ServerCard.Size = UDim2.new(1, -10, 0, 42)
            ServerCard.BackgroundColor3 = Color3.fromRGB(25, 25, 45)
            ServerCard.BackgroundTransparency = 0.3
            ServerCard.BorderSizePixel = 0
            ServerCard.ZIndex = 3

            local CardCorner = Instance.new("UICorner")
            CardCorner.Parent = ServerCard
            CardCorner.CornerRadius = UDim.new(0, 8)

            local InfoText = Instance.new("TextLabel")
            InfoText.Parent = ServerCard
            InfoText.Position = UDim2.new(0, 12, 0, 0)
            InfoText.Size = UDim2.new(0.65, 0, 1, 0)
            InfoText.BackgroundTransparency = 1
            InfoText.Font = Enum.Font.GothamBold
            InfoText.Text = "#" .. i .. " 👥 Jugadores: " .. s.playing .. " / " .. s.maxPlayers
            InfoText.TextColor3 = Color3.fromRGB(220, 220, 255)
            InfoText.TextSize = 12
            InfoText.TextXAlignment = Enum.TextXAlignment.Left
            InfoText.ZIndex = 4

            local JoinCardBtn = Instance.new("TextButton")
            JoinCardBtn.Parent = ServerCard
            JoinCardBtn.Position = UDim2.new(1, -95, 0, 7)
            JoinCardBtn.Size = UDim2.new(0, 85, 0, 28)
            JoinCardBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 100)
            JoinCardBtn.BorderSizePixel = 0
            JoinCardBtn.Font = Enum.Font.GothamBold
            JoinCardBtn.Text = "Unirse"
            JoinCardBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            JoinCardBtn.TextSize = 12
            JoinCardBtn.ZIndex = 4

            local JoinCorner = Instance.new("UICorner")
            JoinCorner.Parent = JoinCardBtn
            JoinCorner.CornerRadius = UDim.new(0, 6)

            JoinCardBtn.MouseButton1Click:Connect(function()
                StatusLabel.Text = "Entrando al servidor #" .. i .. "..."
                StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 255)
                pcall(function()
                    TeleportService:TeleportToPlaceInstance(PlaceId, s.id, LocalPlayer)
                end)
            end)
        end
        
        ListContainer.CanvasSize = UDim2.new(0, 0, 0, #servers * 48)
    else
        StatusLabel.Text = "Buscando servidores..."
        task.wait(1)
        fetchAndDisplayServers()
    end
end

RefreshBtn.MouseButton1Click:Connect(fetchAndDisplayServers)

local autoActive = false
local autoRoutine = nil

AutoBtn.MouseButton1Click:Connect(function()
    autoActive = not autoActive
    if autoActive then
        AutoBtn.Text = "⚡ Auto Hop: ON"
        AutoBtn.BackgroundColor3 = Color3.fromRGB(220, 40, 40)
        autoRoutine = task.spawn(function()
            while autoActive do
                local cards = {}
                for _, child in ipairs(ListContainer:GetChildren()) do
                    if child:IsA("Frame") then
                        table.insert(cards, child)
                    end
                end
                if #cards > 0 then
                    -- Elige de forma aleatoria entre los servidores que están más abajo para evitar el fallo
                    local randomCard = cards[math.random(1, #cards)]
                    for _, obj in ipairs(randomCard:GetChildren()) do
                        if obj:IsA("TextButton") then
                            pcall(function() obj.MouseButton1Click:Fire() end)
                        end
                    end
                else
                    fetchAndDisplayServers()
                end
                task.wait(6)
            end
        end)
    else
        AutoBtn.Text = "⚡ Auto Hop: OFF"
        AutoBtn.BackgroundColor3 = Color3.fromRGB(180, 20, 100)
        if autoRoutine then task.cancel(autoRoutine) end
        StatusLabel.Text = "Auto Hop desactivado."
    end
end)

CloseBtn.MouseButton1Click:Connect(function()
    ScreenGui:Destroy()
end)

task.spawn(fetchAndDisplayServers)
