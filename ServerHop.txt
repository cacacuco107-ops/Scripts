local function serverText(server, index)
    local percent = math.floor(
        (server.playing / server.maxPlayers) * 100
    )

    return string.format(
        "#%d | 👥 %d/%d | %d%% | ID: %s",
        index,
        server.playing,
        server.maxPlayers,
        percent,
        tostring(server.id):sub(1, 8)
    )
end

local function updateServerList()
    local options = {}

    for index, server in ipairs(currentServers) do
        local percent = math.floor(
            (server.playing / server.maxPlayers) * 100
        )

        table.insert(
            options,
            string.format(
                "#%d | 👥 %d/%d | %d%%",
                index,
                server.playing,
                server.maxPlayers,
                percent
            )
        )
    end

    if #options == 0 then
        options = {"No hay servidores disponibles"}
    end

    ServerDropdown:Refresh(options)
    selectedServerIndex = nil
end

--==================================================
-- FETCH SERVERS
--==================================================

local function fetchServers()
    if loading then
        return
    end

    loading = true
    setStatus("🔎 Buscando servidores...")
    ServerCountLabel:Set("Servidores encontrados: ...")

    currentServers = {}

    local success, result = pcall(function()
        return game:HttpGet(
            "https://games.roblox.com/v1/games/"
            .. PlaceId
            .. "/servers/Public?sortOrder=Asc&limit="
            .. MAX_SERVERS
        )
    end)

    if not success or not result then
        setStatus("❌ No se pudo obtener la lista")
        ServerCountLabel:Set("Servidores encontrados: 0")
        loading = false
        return
    end

    local decoded, data = pcall(function()
        return HttpService:JSONDecode(result)
    end)

    if not decoded
        or type(data) ~= "table"
        or type(data.data) ~= "table"
    then
        setStatus("❌ Respuesta inválida")
        ServerCountLabel:Set("Servidores encontrados: 0")
        loading = false
        return
    end

    for _, server in ipairs(data.data) do
        if
            server.id
            and server.id ~= game.JobId
            and type(server.playing) == "number"
            and type(server.maxPlayers) == "number"
            and server.maxPlayers > 0
            and server.playing < server.maxPlayers
        then
            table.insert(currentServers, server)
        end
    end

    table.sort(currentServers, function(a, b)
        local aPercent = a.playing / a.maxPlayers
        local bPercent = b.playing / b.maxPlayers

        if aPercent == bPercent then
            return a.playing < b.playing
        end

        return aPercent < bPercent
    end)

    ServerCountLabel:Set(
        "Servidores encontrados: "
        .. #currentServers
    )

    if #currentServers == 0 then
        setStatus("⚠️ No hay servidores disponibles")
        loading = false
        return
    end

    setStatus(
        "✅ Ordenados por menor ocupación"
    )

    updateServerList()

    loading = false
end

ServerTab:CreateToggle({
    Name = "⚡ AUTO HOP",
    CurrentValue = false,

    Callback = function(value)
        autoHopEnabled = value

        if not value then
            if autoThread then
                task.cancel(autoThread)
                autoThread = nil
            end

            setStatus("Auto Hop: OFF")
            return
        end

        setStatus("⚡ Auto Hop: ON")

        autoThread = task.spawn(function()
            while autoHopEnabled do
                if #currentServers == 0 then
                    fetchServers()
                end

                if #currentServers > 0 then
                    local amount = math.min(
                        LOW_SERVER_POOL,
                        #currentServers
                    )

                    local server = currentServers[
                        math.random(1, amount)
                    ]

                    setStatus(
                        "⚡ Auto Hop → "
                        .. server.playing
                        .. "/"
                        .. server.maxPlayers
                    )

                    pcall(function()
                        TeleportService:TeleportToPlaceInstance(
                            PlaceId,
                            server.id,
                            LocalPlayer
                        )
                    end)

                    task.wait(AUTO_HOP_DELAY)
                else
                    task.wait(3)
                end
            end
        end)
    end
})
