local function clearServerList()

    for _, child in ipairs(
        ServerList:GetChildren()
    ) do

        if child:IsA("Frame") then
            child:Destroy()
        end

    end

    ServerList.CanvasPosition =
        Vector2.new(0, 0)

end

--==================================================
-- SERVER FETCH
--==================================================

local loading = false

local function fetchServers()

    if loading then
        return
    end

    loading = true

    clearServerList()

    ServerCount.Text =
        "SERVIDORES: ..."

    StatusText.Text =
        "🔎 Buscando servidores..."

    StatusText.TextColor3 =
        Color3.fromRGB(255, 210, 100)

    local servers = {}

    local success, result = pcall(function()

        return game:HttpGet(
            "https://games.roblox.com/v1/games/"
            .. PlaceId
            .. "/servers/Public?sortOrder=Asc&limit="
            .. MAX_SERVERS
        )

    end)

    if success and result then

        local decoded, data =
            pcall(function()

                return HttpService:JSONDecode(result)

            end)

        if decoded
            and type(data) == "table"
            and type(data.data) == "table"
        then

            for _, server in ipairs(data.data) do

                if
                    server.id
                    and server.id ~= game.JobId
                    and type(server.playing) == "number"
                    and type(server.maxPlayers) == "number"
                    and server.maxPlayers > 0
                    and server.playing < server.maxPlayers
                then

                    table.insert(
                        servers,
                        server
                    )

                end

            end

        end

    end

    -- Servidores con menos gente primero
    table.sort(
        servers,
        function(a, b)

            local aPercent =
                a.playing / a.maxPlayers

            local bPercent =
                b.playing / b.maxPlayers

            if aPercent == bPercent then
                return a.playing < b.playing
            end

            return aPercent < bPercent

        end
    )

    if #servers == 0 then

        ServerCount.Text =
            "SERVIDORES: 0"

        StatusText.Text =
            "No hay servidores disponibles."

        StatusText.TextColor3 =
            Color3.fromRGB(255, 100, 110)

        loading = false
        return

    end

    ServerCount.Text =
        "SERVIDORES: " .. #servers

    StatusText.Text =
        "✓ Servidores ordenados por ocupación"

    StatusText.TextColor3 =
        Color3.fromRGB(100, 255, 150)

    for index, server in ipairs(servers) do

        createServerCard(
            server,
            index
        )

    end

    task.defer(function()

        ServerList.CanvasSize =
            UDim2.fromOffset(
                0,
                Layout.AbsoluteContentSize.Y + 10
            )

    end)

    loading = false

end

local autoHopEnabled = false
local autoThread = nil

local function setAutoHop(state)

    autoHopEnabled = state

    if state then

        AutoHop.Text =
            "⚡ AUTO: ON"

        AutoHop.BackgroundColor3 =
            Color3.fromRGB(220, 40, 80)

        StatusText.Text =
            "Auto Hop activado."

        StatusText.TextColor3 =
            Color3.fromRGB(255, 150, 180)

    else

        AutoHop.Text =
            "⚡ AUTO: OFF"

        AutoHop.BackgroundColor3 =
            Color3.fromRGB(145, 30, 120)

        StatusText.Text =
            "Auto Hop desactivado."

        StatusText.TextColor3 =
            Color3.fromRGB(190, 200, 225)

    end

end

AutoHop.MouseButton1Click:Connect(function()

    setAutoHop(not autoHopEnabled)

    if not autoHopEnabled then

        if autoThread then

            task.cancel(autoThread)
            autoThread = nil

        end

        return

    end

    autoThread = task.spawn(function()

        while autoHopEnabled do

            local available = {}

            for _, child in ipairs(
                ServerList:GetChildren()
            ) do

                if child:IsA("Frame") then

                    local button =
                        child:FindFirstChildWhichIsA(
                            "TextButton"
                        )

                    if button then
                        table.insert(
                            available,
                            button
                        )
                    end

                end

            end

            if #available == 0 then

                fetchServers()

                task.wait(3)

            else

                local button =
                    available[
                        math.random(
                            1,
                            #available
                        )
                    ]

                if button then

                    pcall(function()
                        button:Activate()
                    end)

                end

                task.wait(6)

            end

        end

    end)

end)

--==================================================
-- CLOSE
--==================================================

Close.MouseButton1Click:Connect(function()

    if autoThread then

        autoHopEnabled = false
        task.cancel(autoThread)
        autoThread = nil

    end

    ScreenGui:Destroy()

end)
